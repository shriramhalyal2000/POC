apiVersion: v1

  name: notebook-secrets

    app: notebook-app

data:

  MONGO_URI: <BASE64_ENCODED_MONGO_URI> 
  
  # Base64 encoded value for the MongoDB Username
  MONGO_USER: <BASE64_ENCODED_MONGO_USERNAME>
  
  # Base64 encoded value for the MongoDB Password

# --- 2. BACKEND Deployment (Port 5000) ---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: notebook-backend-deployment
  labels:
    app: notebook-app
    tier: backend
spec:
  replicas: 1
  selector:
    matchLabels:
      app: notebook-app
      tier: backend
  template:
    metadata:
      labels:
        app: notebook-app
        tier: backend
    spec:
      containers:
      - name: backend
        image: <your-backend-image-name> # REPLACE ME (e.g., myregistry/notebook-backend:v1)
        ports:
        - containerPort: 5000
        env:
          # Inject non-sensitive environment variables from ConfigMap
          - name: APP_VERSION
            valueFrom:
              configMapKeyRef:
                name: notebook-config
                key: APP_VERSION
          # Inject sensitive environment variables from Secret
          - name: MONGO_URI
            valueFrom:
              secretKeyRef:
                name: notebook-secrets
                key: MONGO_URI
          - name: MONGO_USER
            valueFrom:
              secretKeyRef:
                name: notebook-secrets
                key: MONGO_USER
          - name: MONGO_PASS
            valueFrom:
              secretKeyRef:
                name: notebook-secrets
                key: MONGO_PASS
          - name: JWT_SECRET
            valueFrom:
              secretKeyRef:
                name: notebook-secrets
                key: JWT_SECRET
        # Add necessary resource requests/limits for production
        resources:
          limits:
            memory: "128Mi"
            cpu: "500m"

---
# --- 3. BACKEND Service (ClusterIP for Internal Access) ---
# This service is only accessible within the Kubernetes cluster,
# primarily by the 'notebook-frontend-deployment'.
apiVersion: v1
kind: Service
metadata:
  name: notebook-backend-svc
  labels:
    app: notebook-app
    tier: backend
spec:
  selector:
    app: notebook-app
    tier: backend
  ports:
    - protocol: TCP
      port: 5000 # Service Port
      targetPort: 5000 # Container Port
  type: ClusterIP

---

# --- 5. FRONTEND Service (NodePort for External Access via Minikube) ---
# Minikube makes this NodePort accessible from your local machine.
# This Service is used for accessing the login page/UI.
apiVersion: v1
kind: Service
metadata:
  name: notebook-frontend-svc
  labels:
    app: notebook-app
    tier: frontend
spec:
  selector:
    app: notebook-app
    tier: frontend
  ports:
    - protocol: TCP
      port: 80 # Service Port. Requests to the Node's IP:30080 will hit this port.
      targetPort: 3000 # Container Port
      nodePort: 30080 # EXPLICITLY set the port on the node host
  type: NodePort

---
